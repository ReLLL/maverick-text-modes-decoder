cmake_minimum_required(VERSION 3.20)
project(cw_notebook VERSION 1.0.49 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_PLUGIN "Build the host plugin" ON)
set(CW_HOST "brown" CACHE STRING "Plugin host: brown (primary) or sdrpp")
set_property(CACHE CW_HOST PROPERTY STRINGS brown sdrpp)
if(NOT CW_HOST STREQUAL "brown" AND NOT CW_HOST STREQUAL "sdrpp")
  message(FATAL_ERROR "CW_HOST must be brown or sdrpp")
endif()
option(SANITIZE "Instrument standalone tests" OFF)
find_package(Threads REQUIRED)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options("-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-ffile-prefix-map=$ENV{HOME}=~")
endif()
add_subdirectory(vendor)
file(READ "${CMAKE_SOURCE_DIR}/resources/cty.dat" CW_CTY_DATA)
file(READ "${CMAKE_SOURCE_DIR}/resources/country-flags.tsv" CW_FLAG_CODES)
configure_file(src/country_data.h.in generated/country_data.h @ONLY)
add_library(radio_metadata STATIC src/country_lookup.cpp src/radio_text.cpp)
target_include_directories(radio_metadata PUBLIC src PRIVATE "${CMAKE_BINARY_DIR}/generated")
set_target_properties(radio_metadata PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(radio_metadata PRIVATE -Wall -Wextra -Wpedantic)
add_library(cw_core STATIC src/decoder.cpp src/timing_preview.cpp src/timing_estimator.cpp src/journal.cpp src/engine.cpp src/ai_handoff.cpp)
target_include_directories(cw_core PUBLIC src)
target_link_libraries(cw_core PUBLIC Threads::Threads)
set_target_properties(cw_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(cw_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(rtty_core STATIC src/rtty_decoder.cpp src/rtty_skimmer.cpp src/rtty_engine.cpp src/rtty_report.cpp src/psk_decoder.cpp src/psk_monitor.cpp src/fec_monitor.cpp)
target_link_libraries(rtty_core PUBLIC Threads::Threads radio_metadata)
target_include_directories(rtty_core PUBLIC src)
set_target_properties(rtty_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(rtty_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(notebook_fec STATIC src/js8_decoder.cpp src/olivia_decoder.cpp src/fec_worker.cpp)
target_compile_features(notebook_fec PRIVATE cxx_std_20)
target_include_directories(notebook_fec PUBLIC src PRIVATE vendor/gfsk8/src vendor/olivia)
target_link_libraries(notebook_fec PRIVATE notebook_js8 Threads::Threads)
set_target_properties(notebook_fec PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(SANITIZE)
  target_compile_options(notebook_fec PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(notebook_fec PUBLIC -fsanitize=address,undefined)
endif()
target_link_libraries(rtty_core PUBLIC notebook_fec)
if(APPLE)
  target_link_libraries(rtty_core PUBLIC "-framework Accelerate")
endif()
if(SANITIZE)
  target_compile_options(radio_metadata PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(radio_metadata PUBLIC -fsanitize=address,undefined)
  target_compile_options(cw_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cw_core PUBLIC -fsanitize=address,undefined)
  target_compile_options(rtty_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rtty_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(radio_text_tests tests/radio_text_tests.cpp)
target_link_libraries(radio_text_tests PRIVATE radio_metadata)
add_test(NAME radio_text_tests COMMAND radio_text_tests)
add_executable(metadata_tests tests/metadata_tests.cpp)
target_link_libraries(metadata_tests PRIVATE rtty_core)
add_test(NAME metadata_tests COMMAND metadata_tests)
add_executable(snr_activity_tests tests/snr_activity_tests.cpp)
target_link_libraries(snr_activity_tests PRIVATE cw_core rtty_core)
add_test(NAME snr_activity_tests COMMAND snr_activity_tests)
add_executable(olivia_noise_tests tests/olivia_noise_tests.cpp)
target_link_libraries(olivia_noise_tests PRIVATE rtty_core)
add_test(NAME olivia_noise_tests COMMAND olivia_noise_tests)
add_executable(fec_tests tests/fec_tests.cpp)
target_compile_features(fec_tests PRIVATE cxx_std_20)
target_include_directories(fec_tests PRIVATE vendor/olivia)
target_link_libraries(fec_tests PRIVATE notebook_fec notebook_js8 rtty_core)
add_test(NAME fec_tests COMMAND fec_tests)
add_executable(psk_tests tests/psk_tests.cpp)
target_link_libraries(psk_tests PRIVATE rtty_core)
add_test(NAME psk_tests COMMAND psk_tests)
add_executable(rtty_tests tests/rtty_tests.cpp)
target_link_libraries(rtty_tests PRIVATE rtty_core)
add_test(NAME rtty_tests COMMAND rtty_tests)
add_executable(rtty_ui_tests tests/rtty_ui_tests.cpp)
target_link_libraries(rtty_ui_tests PRIVATE rtty_core)
add_test(NAME rtty_ui_tests COMMAND rtty_ui_tests)
add_executable(cw_tests tests/tests.cpp)
target_link_libraries(cw_tests PRIVATE cw_core)
set(CW_RECORDING_FIXTURE "" CACHE FILEPATH "Optional private 8 kHz reception fixture")
if(CW_RECORDING_FIXTURE)
  target_compile_definitions(cw_tests PRIVATE CW_FIXTURE="${CW_RECORDING_FIXTURE}")
endif()
add_test(NAME cw_tests COMMAND cw_tests)
add_executable(cw_recovery_tests tests/recovery_tests.cpp)
target_link_libraries(cw_recovery_tests PRIVATE cw_core)
add_test(NAME cw_recovery_tests COMMAND cw_recovery_tests)
add_executable(cw_acquisition_tests tests/acquisition_tests.cpp)
target_link_libraries(cw_acquisition_tests PRIVATE cw_core)
add_test(NAME cw_acquisition_tests COMMAND cw_acquisition_tests)
add_executable(cw_fading_tests tests/fading_tests.cpp)
target_link_libraries(cw_fading_tests PRIVATE cw_core)
add_test(NAME cw_fading_tests COMMAND cw_fading_tests)
add_executable(cw_noise_tests tests/noise_tests.cpp)
target_link_libraries(cw_noise_tests PRIVATE cw_core)
add_test(NAME cw_noise_tests COMMAND cw_noise_tests)
add_executable(cw_channel_tests tests/channel_tests.cpp)
target_link_libraries(cw_channel_tests PRIVATE cw_core)
add_test(NAME cw_channel_tests COMMAND cw_channel_tests)
add_executable(cw_handoff_tests tests/handoff_tests.cpp)
target_link_libraries(cw_handoff_tests PRIVATE cw_core rtty_core)
add_test(NAME cw_handoff_tests COMMAND cw_handoff_tests)
set(CW_IMGUI_SOURCE "" CACHE PATH "Optional matching host ImGui sources for transcript UI regression tests")
if(CW_IMGUI_SOURCE)
  add_executable(cw_transcript_tests tests/transcript_tests.cpp src/transcript_view.cpp src/transcript_ink.cpp src/notebook_popups.cpp
    "${CW_IMGUI_SOURCE}/imgui.cpp" "${CW_IMGUI_SOURCE}/imgui_draw.cpp"
    "${CW_IMGUI_SOURCE}/imgui_tables.cpp" "${CW_IMGUI_SOURCE}/imgui_widgets.cpp")
  target_include_directories(cw_transcript_tests PRIVATE src "${CW_IMGUI_SOURCE}")
  target_link_libraries(cw_transcript_tests PRIVATE rtty_core)
  add_test(NAME cw_transcript_tests COMMAND cw_transcript_tests)

add_executable(cw_decode tools/decode.cpp)
target_link_libraries(cw_decode PRIVATE cw_core)
if(BUILD_PLUGIN)
  if(NOT APPLE)
    message(FATAL_ERROR "The host plugin currently supports macOS. Use BUILD_PLUGIN=OFF for portable core tests.")
  endif()
  enable_language(OBJCXX)
  set(BROWN_SOURCE "${CMAKE_SOURCE_DIR}/../SDRPlusPlusBrown" CACHE PATH "Brown source headers")
  set(BROWN_APP "/Applications/SDR++Brown.app" CACHE PATH "Installed Brown app")
  if(CW_HOST STREQUAL "brown")
    set(HOST_SOURCE "${BROWN_SOURCE}")
    set(HOST_APP "${BROWN_APP}")
    set(HOST_BROWN 1)
  else()
    set(SDRPP_SOURCE "${CMAKE_SOURCE_DIR}/../SDRPlusPlus" CACHE PATH "Matching SDR++ source headers")
    set(SDRPP_APP "/Applications/SDR++.app" CACHE PATH "Installed SDR++ app")
    set(HOST_SOURCE "${SDRPP_SOURCE}")
    set(HOST_APP "${SDRPP_APP}")
    set(HOST_BROWN 0)
  endif()
  if(NOT EXISTS "${HOST_SOURCE}/core/src/module.h" OR NOT EXISTS "${HOST_APP}/Contents/Frameworks/libsdrpp_core.dylib")
    message(FATAL_ERROR "Set matching source headers and installed app for the selected CW_HOST.")
  endif()
  add_library(cw_notebook MODULE src/plugin.cpp src/plugin_ui.cpp src/plugin_rtty.cpp src/plugin_signals.cpp src/plugin_help.cpp src/plugin_status.cpp src/notebook_popups.cpp src/rtty_receiver.cpp src/plugin_handoff.cpp src/host_adapter.cpp src/transcript_view.cpp src/transcript_ink.cpp src/flag_textures.mm src/plugin_settings.cpp src/plugin_callsigns.cpp)
  target_link_libraries(cw_notebook PRIVATE "-framework AppKit" "-framework OpenGL")
  file(GLOB HOST_VOLK "${HOST_APP}/Contents/Frameworks/libvolk.*.dylib")
  list(LENGTH HOST_VOLK VOLK_COUNT)
  if(NOT VOLK_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected exactly one bundled VOLK library in the selected app")
  endif()
  target_include_directories(cw_notebook SYSTEM PRIVATE "${HOST_SOURCE}/core/src" "${HOST_SOURCE}/core/src/imgui" "${HOST_SOURCE}/decoder_modules/radio/src" /opt/homebrew/include)
  target_compile_definitions(cw_notebook PRIVATE CW_HOST_BROWN=${HOST_BROWN} CW_VERSION_STRING="${PROJECT_VERSION}" CW_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} CW_VERSION_MINOR=${PROJECT_VERSION_MINOR} CW_VERSION_PATCH=${PROJECT_VERSION_PATCH})
  target_link_libraries(cw_notebook PRIVATE cw_core rtty_core "${HOST_APP}/Contents/Frameworks/libsdrpp_core.dylib" "${HOST_VOLK}")
  set_target_properties(cw_notebook PROPERTIES PREFIX "" SUFFIX ".dylib" BUILD_RPATH "${HOST_APP}/Contents/Frameworks")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/cw_notebook.host.json" CONTENT "{\"host\":\"${CW_HOST}\",\"version\":\"${PROJECT_VERSION}\"}\n")
endif()
